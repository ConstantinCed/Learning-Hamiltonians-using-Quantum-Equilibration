\documentclass[tikz,border=4pt]{standalone}

\usepackage{pgfplots}
\usepackage{pgfplotstable}
\usepackage{amsmath,amssymb}
\pgfplotsset{compat=1.18}

\begin{document}

\begin{tikzpicture}

\pgfplotstableread[col sep=comma]{tfim_data_unnormalized_gaussian.csv}\datatable
\pgfplotstableread[col sep=comma]{tfim_gap_fit_gaussian.csv}\fitdata
\pgfplotstablegetelem{0}{C}\of{\fitdata}
\let\fitC\pgfplotsretval
\pgfplotstablegetelem{0}{alpha}\of{\fitdata}
\let\fitAlpha\pgfplotsretval
\pgfplotstablegetelem{0}{beta}\of{\fitdata}
\let\fitBeta\pgfplotsretval
\pgfmathprintnumberto[fixed,precision=2]{\fitAlpha}{\fitAlphaPrint}
\pgfmathprintnumberto[fixed,precision=2]{\fitBeta}{\fitBetaPrint}

\begin{axis}[ymode=log,
    clip mode=individual,
    width=9cm,
    height=7cm,
    xlabel={System size $n$ (qubits)},
    ylabel={Static commutator gap $\pi_H$},
    xmin=4,
    xmax=200,
    grid=both,
    grid style={dashed, gray!30},
    tick align=outside,
    tick style={black},
    label style={font=\small},
    tick label style={font=\small},
    legend style={
        at={(0.97,0.97)},
        anchor=north east,
        draw=none,
        fill=none,
        font=\scriptsize
    }
]

\addplot[
    only marks,
    mark=*,
    mark size=0.8pt,
    color=blue,
    error bars/.cd,
    y dir=both,
    y explicit,
    error bar style={line width=0.6pt},
]
table[
    x=n,
    y=mean,
    y error=sem
] {\datatable};

\addplot[
    red,
    thick,
    dashed,
    domain=4:200,
    samples=300,
]
{\fitC * x^(\fitAlpha) * (ln(x))^(\fitBeta)};

\legend{
TFIM numerics mean $\pm$ SEM,
Fit $n^{\fitAlphaPrint}(\ln n)^{\fitBetaPrint}$
}

\end{axis}
\end{tikzpicture}

\end{document}
