\documentclass[tikz,border=4pt]{standalone}

\usepackage{pgfplots}
\usepackage{pgfplotstable}
\usepackage{amsmath,amssymb}
\pgfplotsset{compat=1.18}

\begin{document}

\begin{tikzpicture}

% If your file is actually named with "(1)", change this line accordingly.
\pgfplotstableread[col sep=comma]{tfim_data_unnormalized_gaussian.csv}\datatable

% ------------------------------------------------------------
% Fit y = C n^alpha (log n)^beta
%
% Linearized form:
%   log(y) = log(C) + alpha log(n) + beta log(log(n))
%
% We skip rows with y <= 0 or n <= 1 to avoid log issues.
% ------------------------------------------------------------

\pgfplotstablegetrowsof{\datatable}
\pgfmathtruncatemacro{\TotalRows}{\pgfplotsretval}
\pgfmathtruncatemacro{\LastRow}{\TotalRows-1}

\pgfmathsetmacro{\Sone}{0}
\pgfmathsetmacro{\Sx}{0}
\pgfmathsetmacro{\Sz}{0}
\pgfmathsetmacro{\Sy}{0}
\pgfmathsetmacro{\Sxx}{0}
\pgfmathsetmacro{\Szz}{0}
\pgfmathsetmacro{\Sxz}{0}
\pgfmathsetmacro{\Sxy}{0}
\pgfmathsetmacro{\Szy}{0}

\pgfplotsforeachungrouped \i in {0,...,\LastRow} {%
    \pgfplotstablegetelem{\i}{n}\of{\datatable}
    \pgfmathsetmacro{\xn}{\pgfplotsretval}

    \pgfplotstablegetelem{\i}{mean}\of{\datatable}
    \pgfmathsetmacro{\yn}{\pgfplotsretval}

    \pgfmathparse{(\xn > 1) && (\yn > 0)}
    \ifnum\pgfmathresult=1
        \pgfmathsetmacro{\lx}{ln(\xn)}
        \pgfmathsetmacro{\lz}{ln(\lx)}
        \pgfmathsetmacro{\ly}{ln(\yn)}

        \pgfmathsetmacro{\SoneNew}{\Sone + 1}
        \pgfmathsetmacro{\SxNew}{\Sx + \lx}
        \pgfmathsetmacro{\SzNew}{\Sz + \lz}
        \pgfmathsetmacro{\SyNew}{\Sy + \ly}
        \pgfmathsetmacro{\SxxNew}{\Sxx + \lx*\lx}
        \pgfmathsetmacro{\SzzNew}{\Szz + \lz*\lz}
        \pgfmathsetmacro{\SxzNew}{\Sxz + \lx*\lz}
        \pgfmathsetmacro{\SxyNew}{\Sxy + \lx*\ly}
        \pgfmathsetmacro{\SzyNew}{\Szy + \lz*\ly}

        \xdef\Sone{\SoneNew}
        \xdef\Sx{\SxNew}
        \xdef\Sz{\SzNew}
        \xdef\Sy{\SyNew}
        \xdef\Sxx{\SxxNew}
        \xdef\Szz{\SzzNew}
        \xdef\Sxz{\SxzNew}
        \xdef\Sxy{\SxyNew}
        \xdef\Szy{\SzyNew}
    \fi
}

% Solve the 3x3 normal equations for
% [a0, a1, a2] = [log(C), alpha, beta]
%
% Matrix:
% [Sone  Sx   Sz ] [a0]   [Sy ]
% [Sx    Sxx  Sxz] [a1] = [Sxy]
% [Sz    Sxz  Szz] [a2]   [Szy]

\pgfmathsetmacro{\DetA}{
    \Sone*(\Sxx*\Szz - \Sxz*\Sxz)
    - \Sx*(\Sx*\Szz - \Sxz*\Sz)
    + \Sz*(\Sx*\Sxz - \Sxx*\Sz)
}

\pgfmathsetmacro{\DetAZero}{
    \Sy*(\Sxx*\Szz - \Sxz*\Sxz)
    - \Sx*(\Sxy*\Szz - \Sxz*\Szy)
    + \Sz*(\Sxy*\Sxz - \Sxx*\Szy)
}

\pgfmathsetmacro{\DetAOne}{
    \Sone*(\Sxy*\Szz - \Sxz*\Szy)
    - \Sy*(\Sx*\Szz - \Sxz*\Sz)
    + \Sz*(\Sx*\Szy - \Sxy*\Sz)
}

\pgfmathsetmacro{\DetATwo}{
    \Sone*(\Sxx*\Szy - \Sxy*\Sxz)
    - \Sx*(\Sx*\Szy - \Sxy*\Sz)
    + \Sy*(\Sx*\Sxz - \Sxx*\Sz)
}

\pgfmathsetmacro{\fitLogC}{\DetAZero/\DetA}
\pgfmathsetmacro{\fitAlpha}{\DetAOne/\DetA}
\pgfmathsetmacro{\fitBeta}{\DetATwo/\DetA}
\pgfmathsetmacro{\fitC}{exp(\fitLogC)}

\pgfmathprintnumberto[fixed,precision=2]{\fitC}{\fitCPrint}
\pgfmathprintnumberto[fixed,precision=2]{\fitAlpha}{\fitAlphaPrint}
\pgfmathprintnumberto[fixed,precision=2]{\fitBeta}{\fitBetaPrint}

\begin{axis}[ymode=log,
    width=9cm,
    height=7cm,
    xlabel={System size $n$ (qubits)},
    ylabel={$\displaystyle \inf_{A\perp H}\frac{1}{d}\|[A,H]\|_F^2$},
    xmin=4,
    xmax=200,
    grid=both,
    grid style={dashed, gray!30},
    tick align=outside,
    tick style={black},
    label style={font=\small},
    tick label style={font=\small},
    legend style={
        at={(0.97,0.97)},
        anchor=north east,
        draw=none,
        fill=none,
        font=\scriptsize
    }
]

\addplot[
    only marks,
    mark=*,
    mark size=0.8pt,
    color=blue,
    error bars/.cd,
    y dir=both,
    y explicit,
    error bar style={line width=0.6pt},
]
table[
    x=n,
    y=mean,
    y error=sem
] {\datatable};

\addplot[
    red,
    thick,
    dashed,
    domain=4:200,
    samples=300,
]
{\fitC * x^(\fitAlpha) * (ln(x))^(\fitBeta)};

\legend{
TFIM numerics mean $\pm$ SEM,
Scaling fit: $\, n^{\fitAlphaPrint} \log^{\fitBetaPrint}n$
}

\end{axis}
\end{tikzpicture}
