% --------------------------------------------------------------------
%  Smoothed-analysis commutator gap for the structured TFIM.
%  Requires: tfim_smoothed_gap.csv (cols: n, eps, median, q25, q75, n_samples)
%  H(eps) = sum_i Z_i Z_{i+1} + eps sum_i g_i X_i ;  g_i ~ N(0, 1).
% --------------------------------------------------------------------
\documentclass[border=4pt]{standalone}

\usepackage{pgfplots}
\usepackage{pgfplotstable}
\usepackage{amsmath,amssymb}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{fillbetween}

\definecolor{royalblue}{HTML}{1f6fb8}
\definecolor{softred}{HTML}{d65c5c}
\definecolor{forestgreen}{HTML}{2c8a3c}

\begin{document}

\pgfplotstableread[col sep=comma]{tfim_smoothed_gap.csv}\gapdata

% Per-n filtered tables. Use \pgfplotstabletypeset-like select via expr.
\pgfplotstableread[col sep=comma]{tfim_smoothed_gap.csv}\gapall

\begin{tikzpicture}
\begin{axis}[
    width=12cm,
    height=7.5cm,
    xlabel={perturbation strength $\varepsilon$  (\,$\varepsilon=0$ shown at left edge\,)},
    ylabel={projected commutator gap $\pi_H(\varepsilon)$},
    xmode=log, log basis x=10,
    ymode=log, log basis y=10,
    xmin=2e-5, xmax=2,
    ymin=1e-13, ymax=1e-1,
    grid=both,
    grid style={line width=.1pt, draw=gray!25},
    major grid style={line width=.2pt, draw=gray!50},
    legend style={font=\small, at={(0.02,0.98)}, anchor=north west,
                  draw=none, fill=white, fill opacity=0.85, text opacity=1},
    title={$H(\varepsilon)=\sum Z_iZ_{i+1}+\varepsilon\sum g_i X_i$,\ $g_i\sim\mathcal{N}(0,1)$,\
           $50$ samples per $\varepsilon>0$},
    title style={font=\small},
]

% ── n = 8 ──
\addplot+[name path=q25_8, draw=none, forget plot]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=q25,
        restrict expr to domain={\thisrow{n}}{8:8}]{\gapall};
\addplot+[name path=q75_8, draw=none, forget plot]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=q75,
        restrict expr to domain={\thisrow{n}}{8:8}]{\gapall};
\addplot[royalblue!25, forget plot] fill between[of=q25_8 and q75_8];
\addplot+[mark=*, mark size=1.6pt, color=royalblue, thick]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=median,
        restrict expr to domain={\thisrow{n}}{8:8}]{\gapall};
\addlegendentry{$n=8$}

% ── n = 10 ──
\addplot+[name path=q25_10, draw=none, forget plot]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=q25,
        restrict expr to domain={\thisrow{n}}{10:10}]{\gapall};
\addplot+[name path=q75_10, draw=none, forget plot]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=q75,
        restrict expr to domain={\thisrow{n}}{10:10}]{\gapall};
\addplot[softred!25, forget plot] fill between[of=q25_10 and q75_10];
\addplot+[mark=square*, mark size=1.5pt, color=softred, thick]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=median,
        restrict expr to domain={\thisrow{n}}{10:10}]{\gapall};
\addlegendentry{$n=10$}

% ── n = 12 ──
\addplot+[name path=q25_12, draw=none, forget plot]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=q25,
        restrict expr to domain={\thisrow{n}}{12:12}]{\gapall};
\addplot+[name path=q75_12, draw=none, forget plot]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=q75,
        restrict expr to domain={\thisrow{n}}{12:12}]{\gapall};
\addplot[forestgreen!25, forget plot] fill between[of=q25_12 and q75_12];
\addplot+[mark=triangle*, mark size=1.8pt, color=forestgreen, thick]
  table[x expr={\thisrow{eps}==0 ? 3e-5 : \thisrow{eps}}, y=median,
        restrict expr to domain={\thisrow{n}}{12:12}]{\gapall};
\addlegendentry{$n=12$}

% Mark eps = 0 column.
\draw[dashed, gray]
  (axis cs:3e-5,1e-13) -- (axis cs:3e-5,1e-1);
\node[anchor=south, gray, font=\scriptsize]
  at (axis cs:3e-5, 1e-13) {$\varepsilon=0$};

\end{axis}
\end{tikzpicture}

\end{document}
