% --------------------------------------------------------------------
%  Smoothed-analysis commutator gap, n = 50, linear axes.
%  Requires: tfim_smoothed_gap.csv (cols: n, eps, median, q25, q75, n_samples)
%  H(eps) = sum_i Z_i Z_{i+1} + eps * sum_i g_i X_i ;  g_i ~ N(0, 1).
% --------------------------------------------------------------------
\documentclass[border=4pt]{standalone}

\usepackage{pgfplots}
\usepackage{pgfplotstable}
\usepackage{amsmath,amssymb}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{fillbetween}

\definecolor{royalblue}{HTML}{1f6fb8}

\begin{document}

\pgfplotstableread[col sep=comma]{tfim_smoothed_gap.csv}\gapdata

\begin{tikzpicture}
\begin{axis}[
    width=12cm,
    height=7.5cm,
    xlabel={perturbation strength $\varepsilon$},
    ylabel={projected commutator gap $\pi_H(\varepsilon)$},
    xmin=-1.05, xmax=1.05,
    ymin=0,
    grid=both,
    grid style={line width=.1pt, draw=gray!25},
    major grid style={line width=.2pt, draw=gray!50},
    legend style={font=\small, at={(0.5,0.98)}, anchor=north,
                  draw=none, fill=white, fill opacity=0.85, text opacity=1},
    title={$H(\varepsilon)=\sum Z_iZ_{i+1}+\varepsilon\sum g_i X_i$,\ $g_i\sim\mathcal{N}(0,1)$,\ $n=50$,\ $30$ samples per $\varepsilon\neq 0$},
    title style={font=\small},
    scaled y ticks=base 10:5,
    ytick scale label code/.code={\pgfmathprintnumber{#1}\,$\times 10^{-5}$},
]

% IQR band
\addplot+[name path=q25, draw=none, forget plot]
  table[x=eps, y=q25]{\gapdata};
\addplot+[name path=q75, draw=none, forget plot]
  table[x=eps, y=q75]{\gapdata};
\addplot[royalblue!25, forget plot] fill between[of=q25 and q75];

% Median
\addplot+[mark=*, mark size=1.5pt, color=royalblue, thick]
  table[x=eps, y=median]{\gapdata};
\addlegendentry{median $\pi_H(\varepsilon)$  (band: 25\%--75\%)}

\draw[dashed, gray] (axis cs:0,\pgfkeysvalueof{/pgfplots/ymin})
                 -- (axis cs:0,\pgfkeysvalueof{/pgfplots/ymax});

\end{axis}
\end{tikzpicture}

\end{document}
